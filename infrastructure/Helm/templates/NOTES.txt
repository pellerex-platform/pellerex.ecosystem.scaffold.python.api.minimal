1. The API listens on container port 8890 (uvicorn) behind a ClusterIP Service on port 80.
{{- if .Values.ingress.enabled }}
2. It is reached through ProxyApi via the ingress paths:
{{- range $host := .Values.ingress.hosts }}
  {{- range .paths }}
  http{{ if $.Values.ingress.tls }}s{{ end }}://{{ $host.host }}{{ .path }}
  {{- end }}
{{- end }}
{{- else }}
2. Port-forward to reach it locally:
  export POD_NAME=$(kubectl get pods --namespace {{ .Release.Namespace }} -l "app.kubernetes.io/name={{ include "RepoUniqueNormalisedIdentifier.name" . }},app.kubernetes.io/instance={{ .Release.Name }}" -o jsonpath="{.items[0].metadata.name}")
  kubectl --namespace {{ .Release.Namespace }} port-forward $POD_NAME 8890:8890
  # then: curl http://127.0.0.1:8890/health/ready
{{- end }}
